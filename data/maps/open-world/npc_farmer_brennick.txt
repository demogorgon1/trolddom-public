# Painted on the routes layer from the cow in the woods to the farmhouse. The trigger numbers are positions along the
# painted route, so they need to match it once it's painted: a wolf at about a quarter, half and three quarters of the way,
# and the event (which completes the objective) where the route reaches the farmhouse.
route .farmer_brennick_cow_escort:
{
	origin_map_entity_spawn: .farmer_brennick_cow

	trigger<2>: { chat: { type: say text: "Moo!" } }
	trigger<16>: { ability: .farmer_brennick_cow_ambush_1 }
	trigger<33>: { ability: .farmer_brennick_cow_ambush_2 }
	trigger<57>: { ability: .farmer_brennick_cow_ambush_3 }
	trigger<63>: { chat: { type: say text: "Mooo!" } event: true }
	trigger<max>: { despawn: true }
}

npc_behavior_state .farmer_brennick_cow:
{
