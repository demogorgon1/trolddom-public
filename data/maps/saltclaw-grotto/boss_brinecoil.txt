# Boss

# Fishing anywhere in this sub zone has a 25% chance per fishing tick to call up Brinecoil. Nothing can be caught
# here (the loot table is empty), so there are no splashes.
zone .brinecoil:
{
	fishing_trigger_ability: .brinecoil_spawn
	fishing_trigger_ability_chance: 25
	fishing_loot_table: $loot_table { }
}

ability .brinecoil_spawn:
{
	direct_effect spawn_entity:
	{
		spawn_flags: [ no_owner ]
		npc_target_threat: 20
		entity: saltclaw_grotto_brinecoil
		map_entity_spawn: .brinecoil
		if_not: .brinecoil_killed
	}
}

encounter .brinecoil:
{
	main_entities: [ saltclaw_grotto_brinecoil ]
}

entity saltclaw_grotto_brinecoil: !NPC
{
	_string: "Brinecoil"
	_level: 11
	_elite: true
	_faction: monsters
	_sprite: tentacle_monster
	_sprite_dead: tentacle_monster_dead
	_weapon_damage: 1.4
	_resource_health: 8
	_loot_table: .brinecoil
	_creature_type: beast
	_encounter: .brinecoil
	_zone: .brinecoil
	_out_of_zone_action: { evade: true }
	_immune_to_stun: true
	_immune_to_immobilize: true
	_immune_to_slow: true
	_can_move_on_all_non_view_blocking_tiles: true
	_evade_despawn: true
	_despawn_time: { never: true }

	_heroic_weapon_damage: 4.0
	_heroic_resource_health: 11
	_loot_table_heroic: .brinecoil_heroic

	_abilities:
	[
		{ id: npc_whirlwind target: self requirement self<must_have_less_health_than>: 90 }
		{ id: npc_frostbolt target: random_player requirement self<must_have_less_health_than>: 95 }
		{ id: npc_attack }
	]
}

map_trigger .brinecoil_killed:
{
	persistent: true
	default: false
}

map_entity_spawn .brinecoil: { spawn_timer: { only_when_triggered: true } }
